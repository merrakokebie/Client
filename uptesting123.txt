using System;
using System.Data.Entity;
using System.Linq;
using System.Web.Mvc;
using YourProject.Models;
using YourProject.ViewModels;

namespace Project.Controllers
{
    [Authorize]
    public class PayPlusApprovalController : Controller
    {
        // Change this if your EF6/EDMX context has a different name.
        private readonly LoanTrackingEntities db = new LoanTrackingEntities();

        [HttpGet]
        public ActionResult Index()
        {
            var centralTimeZone =
                TimeZoneInfo.FindSystemTimeZoneById("Central Standard Time");
            var bankNow =
                TimeZoneInfo.ConvertTimeFromUtc(DateTime.UtcNow, centralTimeZone);

            var today = bankNow.Date;
            var tomorrow = today.AddDays(1);
            var cutoffTime = today.AddHours(17).AddMinutes(10); // 5:10 PM

            // First-pass UI query only.
            // User/client authorization and final status rules come later.
            var wires = db.PayPlusIntakes
                .AsNoTracking()
                .Where(x =>
                    x.IsActive == true &&
                    x.STPEnabled == true &&
                    x.WireDate >= today &&
                    x.WireDate < tomorrow &&
                    x.ApprovedWireTime == null)
                .OrderBy(x => x.WireDate)
                .ThenBy(x => x.PayPlusIntakeId)
                .Select(x => new PayPlusApprovalRowViewModel
                {
                    PayPlusIntakeId = x.PayPlusIntakeId,
                    PayPlusAuthorizedClientId = x.PayPlusAuthorizedClientId,
                    BorrowerFirstName = x.Borrower1FirstName,
                    BorrowerLastName = x.Borrower1LastName,
                    LoanNumber = x.LoanNumber,
                    WireDate = x.WireDate,
                    WireAmount = x.OutgoingWireAmount,
                    NoteAmount = x.NoteAmount,
                    TitleCompanyId = x.TitleCompanyId,
                    SenderReferenceNumber = x.SenderReferenceNumber,

                    // Temporary display value until we join AspNetUsers.
                    InitiatorName = x.CreatedBy,

                    IsRushRequest = x.IsRushRequest
                })
                .ToList();

            var model = new PayPlusApprovalViewModel
            {
                Wires = wires,
                BankNow = bankNow,
                CutoffTime = cutoffTime,
                IsPastCutoff = bankNow > cutoffTime,
                NextBusinessDate = GetNextBusinessDay(today),
                TotalWireAmount = wires.Sum(x => x.WireAmount ?? 0m)
            };

            return View(model);
        }

        private static DateTime GetNextBusinessDay(DateTime date)
        {
            var nextDate = date.AddDays(1);

            while (nextDate.DayOfWeek == DayOfWeek.Saturday ||
                   nextDate.DayOfWeek == DayOfWeek.Sunday)
            {
                nextDate = nextDate.AddDays(1);
            }

            return nextDate;
        }

        protected override void Dispose(bool disposing)
        {
            if (disposing)
            {
                db.Dispose();
            }

            base.Dispose(disposing);
        }
    }
}

using System;
using System.Collections.Generic;

namespace Project.ViewModels
{
    public class PayPlusApprovalViewModel
    {
        public PayPlusApprovalViewModel()
        {
            Wires = new List<PayPlusApprovalRowViewModel>();
        }

        public List<PayPlusApprovalRowViewModel> Wires { get; set; }

        public DateTime BankNow { get; set; }
        public DateTime CutoffTime { get; set; }
        public bool IsPastCutoff { get; set; }
        public DateTime NextBusinessDate { get; set; }
        public decimal TotalWireAmount { get; set; }
    }
    public class PayPlusApprovalRowViewModel
    {
        public int PayPlusIntakeId { get; set; }
        public int? PayPlusAuthorizedClientId { get; set; }

        public string BorrowerFirstName { get; set; }
        public string BorrowerLastName { get; set; }

        public string LoanNumber { get; set; }
        public DateTime? WireDate { get; set; }

        public decimal? WireAmount { get; set; }
        public decimal? NoteAmount { get; set; }

        public int? TitleCompanyId { get; set; }
        public string SenderReferenceNumber { get; set; }
        public string InitiatorName { get; set; }
        public bool IsRushRequest { get; set; }

        // Temporary display helpers until real joins are added.
        public string ClientDisplay
        {
            get
            {
                return PayPlusAuthorizedClientId.HasValue
                    ? "Client " + PayPlusAuthorizedClientId.Value
                    : "-";
            }
        }

        public string BorrowerName
        {
            get
            {
                return ((BorrowerFirstName ?? string.Empty) + " " +
                        (BorrowerLastName ?? string.Empty)).Trim();
            }
        }

        public string TitleCompanyDisplay
        {
            get
            {
                return TitleCompanyId.HasValue
                    ? "Title Co. " + TitleCompanyId.Value
                    : "-";
            }
        }
    }
}

SELECT 'UPDATE CreditLineHistory SET '
     + 'PeriodInterest = '           + ISNULL(CAST(PeriodInterest AS varchar(40)), 'NULL')
     + ', PeriodFee = '              + ISNULL(CAST(PeriodFee AS varchar(40)), 'NULL')
     + ', Balance = '                + ISNULL(CAST(Balance AS varchar(40)), 'NULL')
     + ', PeriodAdjustmentInterest = '+ ISNULL(CAST(PeriodAdjustmentInterest AS varchar(40)), 'NULL')
     + ', Rate = '                   + ISNULL(CAST(Rate AS varchar(40)), 'NULL')
     + ', Fee = '                    + ISNULL(CAST(Fee AS varchar(40)), 'NULL')
     + ', PrimeLIbor = '             + ISNULL(CAST(PrimeLIbor AS varchar(20)), 'NULL')
     + ', PrimePremium = '           + ISNULL(CAST(PrimePremium AS varchar(40)), 'NULL')
     + ', RateFloor = '              + ISNULL(CAST(RateFloor AS varchar(40)), 'NULL')
     + ', NonUsedLimit = '           + ISNULL(CAST(NonUsedLimit AS varchar(40)), 'NULL')
     + ', NonUsedFeeTC = '           + ISNULL(CAST(NonUsedFeeTC AS varchar(40)), 'NULL')
     + ', NonUsedFee = '             + ISNULL(CAST(NonUsedFee AS varchar(40)), 'NULL')
     + ', CurtailmentBalance = '     + ISNULL(CAST(CurtailmentBalance AS varchar(40)), 'NULL')
     + ', CurtailmentInterestMtd = ' + ISNULL(CAST(CurtailmentInterestMtd AS varchar(40)), 'NULL')
     + ', ConstSubLimitUnused = '    + ISNULL(CAST(ConstSubLimitUnused AS varchar(40)), 'NULL')
     + ' WHERE CreditLineID = '      + CAST(CreditLineID AS varchar(20))
     + ' AND PeriodEndDate = '''     + CONVERT(varchar(23), PeriodEndDate, 121) + ''';'
FROM CreditLineHistory
WHERE PeriodEndDate = '2026-08-31'
ORDER BY CreditLineID;

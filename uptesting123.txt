BEGIN TRANSACTION;

-- ── Safety net ────────────────────────────────────────────────
SELECT * INTO LoanHistory_Bak_20260831
FROM LoanHistory WHERE PeriodEndDate = '2026-08-31';

SELECT * INTO CreditLineHistory_Bak_20260831
FROM CreditLineHistory WHERE PeriodEndDate = '2026-08-31';

-- ── Clear Aug 31 ──────────────────────────────────────────────
DELETE FROM LoanHistory       WHERE PeriodEndDate = '2026-08-31';
DELETE FROM CreditLineHistory WHERE PeriodEndDate = '2026-08-31';

-- ── LoanHistory ───────────────────────────────────────────────
INSERT INTO LoanHistory
(
    LoanID, PeriodEndDate, InterestAccrued, CheckFee, DaysOutstanding,
    Over21Fee, WireFee, CheckOnlyFee, AdjustmentFee, AdjustmentInterest, [Keep]
)
SELECT
    LoanID, PeriodEndDate, InterestAccrued, CheckFee, DaysOutstanding,
    Over21Fee, WireFee, CheckOnlyFee, AdjustmentFee, AdjustmentInterest, [Keep]
FROM [LoanTracking_X].[dbo].[LoanHistory]

-- ── CreditLineHistory ─────────────────────────────────────────
INSERT INTO CreditLineHistory
(
    CreditLineID, PeriodEndDate, PeriodInterest, PeriodFee, Balance,
    PeriodAdjustmentInterest, Rate, Fee, PrimeLibor, PrimePremium, RateFloor,
    LineName, LineNumber, DraftAccountNumber, DdaNumber,
    ApprovedAmount, ConstructionSublimit, Line, Over21Fee,
    NonUsedLimit, NonUsedFeeTC, NonUsedFee,
    CurtailmentBalance, CurtailmentInterestMtd, ConstSubLimitUnused

    CreditLineID, PeriodEndDate, PeriodInterest, PeriodFee, Balance,
    PeriodAdjustmentInterest, Rate, Fee, PrimeLibor, PrimePremium, RateFloor,
    LineName, LineNumber, DraftAccountNumber, DdaNumber,
    ApprovedAmount, ConstructionSublimit, Line, Over21Fee,
    NonUsedLimit, NonUsedFeeTC, NonUsedFee,
    CurtailmentBalance, CurtailmentInterestMtd, ConstSubLimitUnused

-- ── Verify counts match source ────────────────────────────────
SELECT 'LoanHistory' AS TableName,
       (SELECT COUNT(*) FROM [LoanTracking_X].[dbo].[LoanHistory]
        WHERE PeriodEndDate = '2026-08-31') AS SourceRows,
       (SELECT COUNT(*) FROM LoanHistory
        WHERE PeriodEndDate = '2026-08-31') AS TargetRows
UNION ALL
SELECT 'CreditLineHistory',
       (SELECT COUNT(*) FROM [LoanTracking_X].[dbo].[CreditLineHistory]
        WHERE PeriodEndDate = '2026-08-31'),
       (SELECT COUNT(*) FROM CreditLineHistory
        WHERE PeriodEndDate = '2026-08-31');

-- ── Fee sanity check — should be non-zero ─────────────────────
